from openai import OpenAI
c = OpenAI(base_url="http://127.0.0.1:8787/v1", api_key="onyx")
print(c.chat.completions.create(model="onyx-v0.5",
      messages=[{"role":"user","content":"Привет"}], max_tokens=64
).choices[0].message.content)
JavaScript 
const r = await fetch("http://127.0.0.1:8787/v1/chat/completions", {
  method: "POST", headers: {"Content-Type": "application/json"},
  body: JSON.stringify({model: "onyx-v0.5",
    messages: [{role: "user", content: "Привет"}], max_tokens: 64})
});
console.log((await r.json()).choices[0].message.content);
